# /etc/systemd/system/aliza-dashboard.service
[Unit]
Description=AlizaAI Dashboard API
After=network.target

[Service]
Type=simple
User=ubuntu
WorkingDirectory=/opt/aliza-ai
EnvironmentFile=-/opt/aliza-ai/.env
ExecStart=/opt/aliza-ai/venv/bin/python /opt/aliza-ai/scripts/run_dashboard.py
Restart=always
RestartSec=5

[Install]
WantedBy=multi-user.target

[Service]
UMask=0077
